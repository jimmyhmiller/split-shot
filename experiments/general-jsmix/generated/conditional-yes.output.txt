export function main(program) {
  "use strict";
  let a, b, c, d, effects, condition, yesEffects, noEffects, yesValue, noValue, g, h, right;
  a = codegen.buffer();
  b = codegen.reference("io");
  h = codegen.call(a, b, "prompt", []);
  right = program[0];
  c = codegen.binary("===", h, right);
  d = codegen.fork(a);
  effects = a;
  condition = c;
  yesEffects = d;
  noEffects = codegen.fork(a);
  try {
    b = codegen.reference("io");
    yesValue = codegen.call(yesEffects, b, "prompt", []);
  } catch (e) {
    yesValue = codegen.caught(yesEffects, e)[1];
  }
  try {
    b = codegen.reference("io");
    noValue = codegen.call(noEffects, b, "prompt", []);
  } catch (f) {
    noValue = codegen.caught(noEffects, f)[1];
  }
  g = codegen.conditional(effects, condition, yesEffects, yesValue, noEffects, noValue);
  return codegen.finish(a, g);
}
